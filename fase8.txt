-- SUBSTR(coluna, início, tamanho)
SELECT
    nm_cidadao, 
    SUBSTR(cd_cpf_cidadao,1,3) || '.' ||
    SUBSTR(cd_cpf_cidadao,4,3) || '.' ||
    SUBSTR(cd_cpf_cidadao,7,3) || '-' ||
    SUBSTR(cd_cpf_cidadao,10,2) AS cpf_formatado
FROM CIDADAO;

SELECT UPPER(nm_cidadao) FROM CIDADAO;

SELECT 
    nm_cidadao,
    TRUNC(MONTHS_BETWEEN(SYSDATE, dt_nascimento)/12) AS idade
FROM CIDADAO;

SELECT 
    id_ocorrencia,
    TO_CHAR(dt_ocorrencia, 'MM') AS mes,
    TO_CHAR(dt_ocorrencia, 'YYYY') AS ano
FROM OCORRENCIA;
