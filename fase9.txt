-- Total de ocorrências por serviço

SELECT s.nm_servico, COUNT(*) AS total_ocorrencias
FROM ocorrencia o JOIN servico_publico s
ON o.id_servico = s.id_servico
GROUP BY s.nm_servico;

-- exibir apenas seriços com mais de X ocorrencias

SELECT s.nm_servico, COUNT(*) AS total_ocorrencias
FROM ocorrencia o JOIN servico_publico s
ON o.id_servico = s.id_servico
GROUP BY s.nm_servico HAVING COUNT(*) > 1;
